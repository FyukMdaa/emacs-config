* setup.el のブートストラップ

twist.nix がすでにすべてのパッケージを Nix store 上にビルド・配置して
いるので、setup.el 標準の =:package= は何もしない =no-op= に置き換える。

#+begin_src emacs-lisp
;; -*- lexical-binding: t; -*-

(eval-and-compile
  (require 'setup))

(setup-define :package
  (lambda (_order) nil)
  :documentation "no-op: パッケージのインストールは twist.nix が担当する。"
  :repeatable t
  :shorthand #'cadr)
#+end_src

** =setopt= を =:opt= で使えるようにする
#+begin_src emacs-lisp
(setup-define :opt
  (lambda (&rest pairs)
    `(setopt ,@pairs))
  :after-loaded t)
#+end_src
** =:load-after= を使えるようにする
#+begin_src emacs-lisp
  (setup-define :load-after
  (lambda (&rest features)
    (let ((body `(require ',(setup-get 'feature))))
      (dolist (feature (nreverse features))
        (setq body `(with-eval-after-load ',feature ,body)))
      body))
  :documentation "Load the current feature after FEATURES.")
#+end_src
** =:needs= を使えるようにする
#+begin_src emacs-lisp
  (setup-define :needs
  (lambda (executable)
    `(unless (executable-find ,executable)
       ,(setup-quit)))
  :documentation "If EXECUTABLE is not in the path, stop here."
  :repeatable 1)
#+end_src

** =:mode-remap= を使えるようにする
#+begin_src emacs-lisp
    (setup-define :mode-remap
    (lambda (old-mode)
      `(add-to-list 'major-mode-remap-alist
                     (cons ',old-mode ',(setup-get 'mode))))
    :documentation "Remap OLD-MODE to the current mode via `major-mode-remap-alist'.
  This causes Emacs to automatically use the current mode instead of
  OLD-MODE whenever OLD-MODE would otherwise be activated (e.g. by
  `auto-mode-alist' or `set-auto-mode'). Useful for redirecting legacy
  major modes to their tree-sitter (or other) replacements without
  touching `auto-mode-alist' itself."
    :repeatable t
    :signature '(OLD-MODE ...))

#+begin_src emacs-lisp
  (setup (:package treemacs)
    (:global-bind "C-c t" treemacs)
    (:bind [mouse-1] treemacs-single-click-expand-action)
    (:option treemacs-text-scale -1)
    (:option
     treemacs-last-error-persist-file (expand-file-name "~/emacs-twist/treemacs/persist-last-error.org")
     treemacs-persist-file (expand-file-name "~/emacs-twist/treemacs/persist.org"))
    ;; VSCode のようにシングルクリックでファイル展開しつつフォーカスは移さない。
    (:when-loaded
      (defun my-treemacs-expand-action (&rest _)
        "Keep the focus on the Treemacs buffer after an action."
        (select-window (treemacs-get-local-window)))
      (advice-add 'treemacs-single-click-expand-action :after #'my-treemacs-expand-action)
        ;; modeline に mlscroll を載せる。
      (defun my-treemacs-setup-mode-line ()
        "Configure a Treemacs-specific mode line with moody and mlscroll."
        (setq-local mlscroll-right-align nil)
        (setq-local mode-line-format
                    '("%e"
                      (:eval (moody-tab " Treemacs " 10 'down))
                      mode-line-format-right-align
                      (:eval (mlscroll-mode-line)
                             " ")))
        (force-mode-line-update))
      
      (add-hook 'treemacs-mode-hook #'my-treemacs-setup-mode-line)
      ;; 右フリンジの継続と切り詰めの矢印表示を無効にする。
      (defun my-treemacs-hide-fringe-indicators ()
        "Hide continuation and truncation fringe indicators in Treemacs."
        (setq-local fringe-indicator-alist
                    (let ((alist (copy-tree fringe-indicator-alist)))
                      (setf (cdr (assq 'truncation alist)) '(nil nil)
                            (cdr (assq 'continuation alist)) '(nil nil))
                      alist)))
      
      (add-hook 'treemacs-mode-hook #'my-treemacs-hide-fringe-indicators)))
  
  (setup (:package treemacs-nerd-icons)
    (:load-after treemacs nerd-icons)
    (:require treemacs-nerd-icons)
    (treemacs-load-theme "nerd-icons"))
#+end_src
