* setup.el のブートストラップ

twist.nix がすでにすべてのパッケージを Nix store 上にビルド・配置して
いるので、setup.el 標準の =:package= は何もしない =no-op= に置き換える。
#+begin_src emacs-lisp
;; -*- lexical-binding: t; -*-

(eval-and-compile
  (require 'setup))

(setup-define :package
  (lambda (_order) nil)
  :documentation "no-op: パッケージのインストールは twist.nix が担当する。"
  :repeatable t
  :shorthand #'cadr)
#+end_src

** =:load-after= を使えるようにする
#+begin_src emacs-lisp
  (setup-define :load-after
  (lambda (&rest features)
    (let ((body `(require ',(setup-get 'feature))))
      (dolist (feature (nreverse features))
        (setq body `(with-eval-after-load ',feature ,body)))
      body))
  :documentation "Load the current feature after FEATURES.")
#+end_src

*** moody

#+begin_src emacs-lisp
(setup (:package moody)
  (:opt x-underline-at-descent-line t)
  (:require moody)
  (moody-replace-mode-line-buffer-identification)
  (moody-replace-eldoc-minibuffer-message-function)

  ; dimmerを効かせる
  (defun my-dimmer-background-active-p ()
    "Return non-nil when `dimmer' is active for background dimming."
  (and (bound-and-true-p dimmer-mode)
       (boundp 'dimmer-buffer-face-remaps)
       dimmer-buffer-face-remaps
       (eq dimmer-adjustment-mode :background)
       (fboundp 'dimmer-cached-compute-rgb)))

  (defun my-dim-background-color (color)
    "Return COLOR dimmed with the same background adjustment as `dimmer'."
    (when (and color
               (stringp color)
               (my-dimmer-background-active-p)
               (color-defined-p color)
               (color-defined-p (face-foreground 'default)))
      (dimmer-cached-compute-rgb color
                                 (face-foreground 'default)
                                 dimmer-fraction
                                 dimmer-use-colorspace)))

  (defun my-moody-dim-inactive-face-attribute (orig-fn face attribute &optional frame inherit)
    "Return dimmed Moody colors for inactive tabs."
    (let ((value (funcall orig-fn face attribute frame inherit)))
      (pcase (list face attribute)
        (`(default :background)
         (or (my-dim-background-color value) value))
        (`(mode-line-inactive :background)
         (or (my-dim-background-color value) value))
        (`(mode-line-inactive :underline)
         (let* ((bg (funcall orig-fn 'mode-line-inactive :background frame inherit))
                (line (cond
                       ((and value (listp value)) (plist-get value :color))
                       ((eq value 'unspecified) bg)
                       ((or value bg))))
                (dimmed (and line (my-dim-background-color line))))
           (cond
            ((and dimmed (listp value)) (plist-put (copy-sequence value) :color dimmed))
            (dimmed dimmed)
            (t value))))
        (_ value))))

  (defun my-moody-dim-inactive-tab-a (orig-fn &rest args)
    "Dim inactive Moody separators with the same algorithm as `dimmer'."
    (if (or (moody-window-active-p)
            (not (my-dimmer-background-active-p)))
        (apply orig-fn args)
      (let ((orig-face-attribute (symbol-function 'face-attribute)))
        (cl-letf (((symbol-function 'face-attribute)
                   (lambda (face attribute &optional frame inherit)
                     (my-moody-dim-inactive-face-attribute
                      orig-face-attribute face attribute frame inherit))))
          (apply orig-fn args)))))

  (advice-add 'moody-wrap :around #'my-moody-dim-inactive-tab-a)
  (moody-replace-mode-line-buffer-identification)
  ;; (moody-replace-vc-mode)
  (moody-replace-eldoc-minibuffer-message-function))
